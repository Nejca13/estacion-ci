Ejercicio 5 – Punteros a struct vs paso por valor

¿Qué imprime cada programa? Explicar:

/* A */
typedef struct { float v; } S;
void f1(S s) { s.v = 10; }
int main() { S x={5}; f1(x); printf("A: %.0f\n", x.v); }

/* B */
void f2(S *s) { s->v = 10; }
int main() { S x={5}; f2(&x); printf("B: %.0f\n", x.v); }

/* C */
void f3(S *s) { s = NULL; }  // ¿Cambia el original?
int main() { S x={5}; f3(&x); printf("C: %.0f\n", x.v); }

/* D */
void f4(S **s) { *s = NULL; }
int main() { S x={5}; S *p=&x; f4(&p); printf("D: p=%p\n", (void*)p); }
