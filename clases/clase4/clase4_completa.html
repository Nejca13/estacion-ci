#include 
#include 

// Definición del struct (fuera de main, global)
typedef enum {
    TIPO_DIGITAL = 0,   // DHT11, RTC
    TIPO_ANALOGICO = 1  // LM35, LDR, Water
} TipoSensor;

typedef struct {
    char nombre[20];        // "DHT11", "LM35", "LDR", "LLUVIA"
    uint8_t pin;            // Pin Arduino (0-255)
    TipoSensor tipo;        // DIGITAL o ANALOGICO
    float valor;            // Última lectura
    float umbral_alerta;    // Umbral para alerta
    bool activo;            // Si el sensor está habilitado
    float offset_calibracion; // Corrección (ej: +0.3°C)
} Sensor;

// Ejemplo de inicialización
Sensor dht11 = {
    .nombre = "DHT11_TEMP",
    .pin = 2,
    .tipo = TIPO_DIGITAL,
    .valor = 0.0,
    .umbral_alerta = 30.0,
    .activo = true,
    .offset_calibracion = 0.0
};
💡 typedef enum: Nombres legibles (TIPO_DIGITAL) en vez de números mágicos. uint8_t para pin (1 byte, 0-255).
💡 En Arduino (.ino) es idéntico: struct funciona igual en C y en Arduino C++. Pegá este mismo código en tu sketch de Clase 3 y compila.

3. Funciones que operan sobre Sensor (paso por VALOR - copia)

Si pasamos el struct por valor, la función recibe una copia. Los cambios NO afectan el original:

#include 

typedef struct {
    char nombre[20];
    float valor;
    float offset;
} Sensor;

// Paso por VALOR: recibe COPIA
void aplicar_offset_valor(Sensor s) {
    s.valor += s.offset;  // Modifica la copia
    printf("  Dentro: %s = %.2f (offset aplicado)\n", s.nombre, s.valor);
}

int main() {
    Sensor lm35 = { .nombre="LM35", .valor=25.0, .offset=0.3 };

    printf("Antes: %s = %.1f\n", lm35.nombre, lm35.valor);  // 25.0
    aplicar_offset_valor(lm35);                              // Copia
    printf("Después: %s = %.1f\n", lm35.valor);              // 25.0 ← NO CAMBIÓ!

    return 0;
}
⚠️ Problema: Pasar struct por valor copia TODOS los bytes (lento, gasta stack). En embebidos con structs grandes = malo. Y no modifica el original.

Bloque 2 – Punteros: &, *, paso por referencia

Modificar sensores sin copiar, funciones que cambian el original

1. Punteros: dirección y valor

Un puntero es una variable que guarda la dirección de memoria de otra variable. En embebidos es esencial: permite modificar el original sin copiar, y acceder a hardware (registros).

#include 

int main() {
    float temp = 24.5;
    float *ptr = &temp;   // & = "dirección de" → ptr guarda dónde está temp

    printf("temp  = %.1f\n", temp);       // 24.5
    printf("&temp = %p\n", (void*)&temp); // dirección en hex (ej: 0x7ffd...)
    printf("ptr   = %p\n", (void*)ptr);   // misma dirección
    printf("*ptr  = %.1f\n", *ptr);        // 24.5  → * = "valor en esa dirección"

    // Modificar vía puntero
    *ptr = 30.0;
    printf("temp ahora = %.1f\n", temp);   // 30.0 ← CAMBIÓ!

    return 0;
}
Operador Nombre Ejemplo
&Dirección deptr = &temp
*Desreferenciar (valor en)val = *ptr
->Miembro vía punteroptr->valor = (*ptr).valor

2. Paso por referencia: funciones que MODIFICAN el sensor

En lugar de pasar el struct (copia), pasamos su dirección (Sensor*). La función recibe la dirección y usa -> para modificar el original:

#include 
#include 

typedef struct {
    char nombre[20];
    float valor;
    float offset;
    bool activo;
} Sensor;

// Paso por REFERENCIA: recibe DIRECCIÓN (Sensor*)
void aplicar_offset(Sensor *s) {
    if (!s->activo) return;          // -> = (*s).activo
    s->valor += s->offset;           // Modifica EL ORIGINAL
    printf("  [FUNCIÓN] %s: %.2f + %.2f = %.2f\n", s->nombre, s->valor - s->offset, s->offset, s->valor);
}

void activar_sensor(Sensor *s) {
    s->activo = true;
}

void desactivar_sensor(Sensor *s) {
    s->activo = false;
}

int main() {
    Sensor lm35 = { .nombre="LM35", .valor=25.0, .offset=0.3, .activo=true };

    printf("Antes: %s = %.1f (activo=%d)\n", lm35.nombre, lm35.valor, lm35.activo);

    // Pasamos DIRECCIÓN con &
    aplicar_offset(&lm35);           // & = dirección de lm35

    printf("Después: %s = %.1f\n", lm35.nombre, lm35.valor);  // 25.3 ← CAMBIÓ!

    // Desactivar
    desactivar_sensor(&lm35);
    printf("Activo: %d\n", lm35.activo);  // 0

    return 0;
}
Regla de oro: &variable al llamar, Tipo* parametro en la función, ptr->campo dentro. Siempre pasar structs por puntero en embebidos.

Leer sensor y guardar en struct (Arduino real + simulación)

#include <DHT.h>
#define DHT_PIN 2
#define DHT_TYPE DHT11
DHT dht(DHT_PIN, DHT_TYPE);

// #define SIMULACION  // ← descomentar si compiles en PC sin placa

void leer_sensor(Sensor *s) {
    if (!s->activo) return;

    if (s->tipo == TIPO_ANALOGICO) {
#ifdef SIMULACION
        int raw = 400 + rand() % 300;  // PC sin hardware
#else
        int raw = analogRead(s->pin);   // Arduino real: LM35(A0), LDR(A1)
#endif
        if (strcmp(s->nombre, "LM35") == 0) {
            s->valor = (raw * 5.0 / 1023.0) * 100.0 + s->offset;  // °C (Clase 1 fórmula)
        } else {
            s->valor = raw;  // LDR/LLUVIA: 0-1023
        }
    } else { // DIGITAL
        if (strcmp(s->nombre, "DHT11_TEMP") == 0) {
#ifdef SIMULACION
            s->valor = 24.5;
#else
            float t = dht.readTemperature();
            s->valor = isnan(t) ? s->valor : t; // mantiene último si falla (Clase 3)
#endif
        } else if (strcmp(s->nombre, "DHT11_HUM") == 0) {
#ifdef SIMULACION
            s->valor = 62.0;
#else
            float h = dht.readHumidity();
            s->valor = isnan(h) ? s->valor : h;
#endif
        }
    }
#ifdef SIMULACION
    printf("  [LECTURA] %s: %.1f\n", s->nombre, s->valor);
#else
    Serial.print("  [LECTURA] "); Serial.print(s->nombre); Serial.print(": "); Serial.println(s->valor,1);
#endif
}

Aplicar calibración (offset) — mismo en PC y Arduino

void calibrar(Sensor *s) {
    if (s->offset != 0.0) {
        s->valor += s->offset;
#ifdef SIMULACION
        printf("  [CAL] %s: offset %.2f → %.2f\n", s->nombre, s->offset, s->valor);
#else
        Serial.print("  [CAL] "); Serial.print(s->nombre); Serial.println(s->valor,1);
#endif
    }
}

Verificar alertas (reusa if de Clase 2 + LEDs Arduino)

void verificar_alertas(Sensor *s) {
    if (!s->activo) return;
    bool critico = s->valor > s->umbral_critico;
    bool alerta  = s->valor > s->umbral_alerta;

    if (strcmp(s->nombre, "LLUVIA") == 0 && alerta) {
#ifdef SIMULACION
        printf("  🌧️ ALERTA %s: LLUVIA (%.0f)\n", s->nombre, s->valor);
#else
        Serial.print("🌧️ LLUVIA "); Serial.println(s->valor,0);
        digitalWrite(10, HIGH); // LED lluvia
#endif
    } else if (critico) {
#ifdef SIMULACION
        printf("  🔴 CRÍTICO %s: %.1f > %.1f\n", s->nombre, s->valor, s->umbral_critico);
#else
        digitalWrite(8, HIGH); // LED rojo - lógica Clase 2
#endif
    } else if (alerta) {
#ifdef SIMULACION
        printf("  🟠 ALERTA %s: %.1f > %.1f\n", s->nombre, s->valor, s->umbral_alerta);
#else
        digitalWrite(9, HIGH); // LED naranja
#endif
    }
}

1. Funciones que recorren TODO el array (reutilizan Bloque 2)

En lugar de repetir código, delegamos en leer_sensor(&s) del Bloque 2 (ya resuelve DHT/analogRead):

#include <stdio.h>
#include <string.h>
// En Arduino: #include <DHT.h> + DHT dht(2,DHT11);

// Leer TODOS (for + continue de Clase 3)
void leer_todos(Sensor sensores[], int total) {
    for (int i = 0; i < total; i++) {
        if (!sensores[i].activo) continue; // salta desactivos
        leer_sensor(&sensores[i]);         // puntero + función del Bloque 2
    }
}
void calibrar_todos(Sensor sensores[], int total) {
    for (int i = 0; i < total; i++) if(sensores[i].activo) calibrar(&sensores[i]);
}
void alertar_todos(Sensor sensores[], int total) {
    for (int i = 0; i < total; i++) if(sensores[i].activo) verificar_alertas(&sensores[i]);
}
// Búsqueda lineal (for + strcmp)
Sensor* buscar_sensor(Sensor sensores[], int total, const char *nombre) {
    for (int i = 0; i < total; i++) if(strcmp(sensores[i].nombre, nombre)==0) return &sensores[i];
    return NULL;
}

void menu_actualizar(Sensor sensores[], int total, int *idx, bool *en_detalle){
    static unsigned long ult_btn=0; if(millis()-ult_btn<200) return;
    int btn_m = digitalRead(BTN_MENU); // LOW=presionado
    int btn_s = digitalRead(BTN_SEL);
    static int m_ant=HIGH, s_ant=HIGH;

3. Sistema completo: loop() no bloqueante (integra Clases 2 + 3 + hoy)

Un loop() que con millis() (Clase 3) hace: leer → alertar (if Clase2) → enviar CSV al dashboard → atender botones/LCD.

#include <DHT.h>
#define DHT_PIN 2
#define DHT_TYPE DHT11
DHT dht(DHT_PIN, DHT_TYPE);
#define BTN_MENU 3
#define BTN_SEL  4
#define INTERVALO_MS 2000

Sensor sensores[6]; // se llena en setup() con init_sensores()
unsigned long ultima=0; int idx_menu=0; bool en_detalle=false;

void setup(){
    Serial.begin(9600); dht.begin();
    pinMode(BTN_MENU, INPUT_PULLUP); pinMode(BTN_SEL, INPUT_PULLUP);
    pinMode(8,OUTPUT); pinMode(9,OUTPUT); pinMode(10,OUTPUT);
    init_sensores(sensores); // Bloque 1
    Serial.println("ESTACION_METEO_CSV"); // cabecera que espera el dashboard Clase 3
}
void loop(){
    unsigned long ahora=millis();
    // 1. LECTURA PERIÓDICA no bloqueante (Clase 3)
    if(ahora - ultima >= INTERVALO_MS){
        leer_todos(sensores,6);      // for + punteros (hoy)
        alertar_todos(sensores,6);   // if/switch (Clase2)
        // 2. ENVIAR CSV al dashboard (Clase 3 formato: temp,hum)
        // Para compatibilidad: primero versión simple temp,hum
        Sensor *t=buscar_sensor(sensores,6,"DHT11_TEMP");
        Sensor *h=buscar_sensor(sensores,6,"DHT11_HUM");
        if(t && h){ Serial.print(t->valor,1); Serial.print(","); Serial.println(h->valor,0); }
        // + opcional: dump completo para debug
        // for(int i=0;i<6;i++) {Serial.print(sensores[i].nombre);Serial.print(",");Serial.println(sensores[i].valor,1);}
        ultima=ahora;
    }
    // 3. BOTONES + LCD (no bloqueante, 200ms debounce)
    menu_actualizar(sensores,6,&idx_menu,&en_detalle);
    // 4. COMANDOS desde PC/dashboard: 'R'=lectura inmediata, 'E'=eco (Clase 3)
    if(Serial.available()){ char c=Serial.read(); if(c=='R'){ Sensor *t=buscar_sensor(sensores,6,"LM35"); if(t){ leer_sensor(t); Serial.print("AHORA ");Serial.print(t->nombre);Serial.print(",");Serial.println(t->valor,1);} } }
}

void init_sensores(Sensor s[]) {
    // 1. DHT11_TEMP
    strcpy(s[0].nombre, "DHT11_TEMP"); s[0].pin=2; s[0].tipo=TIPO_DIGITAL;
    s[0].valor=0; s[0].umbral_alerta=UMBRAL_TEMP_ALERTA; s[0].umbral_critico=UMBRAL_TEMP_CRITICO;
    s[0].activo=true; s[0].offset=0.0;

    // 2. DHT11_HUM
    strcpy(s[1].nombre, "DHT11_HUM"); s[1].pin=2; s[1].tipo=TIPO_DIGITAL;
    s[1].valor=0; s[1].umbral_alerta=UMBRAL_HUM_ALERTA; s[1].umbral_critico=UMBRAL_HUM_CRITICO;
    s[1].activo=true; s[1].offset=0.0;

    // 3. LM35
    strcpy(s[2].nombre, "LM35"); s[2].pin=14; s[2].tipo=TIPO_ANALOGICO;  // A0=14
    s[2].valor=0; s[2].umbral_alerta=UMBRAL_TEMP_ALERTA; s[2].umbral_critico=UMBRAL_TEMP_CRITICO;
    s[2].activo=true; s[2].offset=0.3;  // calibración

    // 4. LDR
    strcpy(s[3].nombre, "LDR"); s[3].pin=15; s[3].tipo=TIPO_ANALOGICO;  // A1=15
    s[3].valor=0; s[3].umbral_alerta=UMBRAL_LUZ_ALERTA; s[3].umbral_critico=UMBRAL_LUZ_CRITICO;
    s[3].activo=true; s[3].offset=0.0;

    // 5. LLUVIA
    strcpy(s[4].nombre, "LLUVIA"); s[4].pin=16; s[4].tipo=TIPO_ANALOGICO;  // A2=16
    s[4].valor=0; s[4].umbral_alerta=UMBRAL_LLUVIA_ALERTA; s[4].umbral_critico=UMBRAL_LLUVIA_CRITICO;
    s[4].activo=true; s[4].offset=0.0;

    // 6. RTC (solo hora, sin alertas)
    strcpy(s[5].nombre, "RTC"); s[5].pin=5; s[5].tipo=TIPO_DIGITAL;
    s[5].valor=0; s[5].umbral_alerta=0; s[5].umbral_critico=0;
    s[5].activo=true; s[5].offset=0.0;
}

Detalles de leer_sensor (simulación según tipo)

void leer_sensor(Sensor *s) {
    if (!s->activo) return;

    if (s->tipo == TIPO_ANALOGICO) {
        int raw = 400 + rand() % 300;  // 400-700 simulado
        if (strcmp(s->nombre, "LM35") == 0) {
            s->valor = (raw * 5.0 / 1023.0) * 100.0 + s->offset;  // °C
        } else {
            s->valor = raw;  // LDR, LLUVIA: raw 0-1023
        }
    } else {  // DIGITAL
        if (strcmp(s->nombre, "DHT11_TEMP") == 0) s->valor = 24.5;
        else if (strcmp(s->nombre, "DHT11_HUM") == 0) s->valor = 62.0;
        else s->valor = 0;  // RTC
    }
}
